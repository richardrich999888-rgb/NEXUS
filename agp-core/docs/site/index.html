Quick Start

# Install AGP-OS
pip install agp-os

# Import and initialize
from agp_os import GovernanceClient, TELOSMembrane, AHES

# Create governance client
client = GovernanceClient(api_key="your-api-key")

# Evaluate an agent action
result = client.evaluate_action(
    agent_id="agent-1",
    action="send_transaction",
    context={"amount": 1000, "recipient": "0x..."}
)

if result.allowed:
    print("Action approved with alignment:", result.alignment_score)
else:
    print("Action blocked:", result.reason)
