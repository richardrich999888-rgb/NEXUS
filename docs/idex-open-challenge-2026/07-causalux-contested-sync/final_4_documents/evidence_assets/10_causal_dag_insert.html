  90      /// Insert an operation into the DAG
  91      pub fn insert(&mut self, op: CausalOp) -> Result<(), DagError> {
  92          // 1. Idempotence check
  93          if self.operations.contains_key(&op.id) {
  94              return Ok(());
  95          }
  96  
  97          // 2. Validate dependencies exist
  98          for dep_id in &op.dependencies {
  99              if !self.operations.contains_key(dep_id) {
 100                  return Err(DagError::MissingDependency(dep_id.clone()));
 101              }
 102          }
 103  
 104          // 3. Check for conflicts with existing operations
 105          let mut merged_results = Vec::new();
 106          for existing_op in self.operations.values() {
 107              if op.conflicts_with(existing_op) {
 108                  match self.conflict_resolver.resolve(&op, existing_op) {
 109                      ConflictResolution::NoConflict => {}
 110                      ConflictResolution::Winner(winner_id) => {
 111                          if winner_id != op.id {
 112                              return Err(DagError::ConflictRejected { winner_id });
 113                          }
 114                      }
 115                      ConflictResolution::Manual { option_a, option_b } => {
 116                          return Err(DagError::ManualResolutionRequired {
 117                              op_a: option_a.id,
 118                              op_b: option_b.id,
 119                          });
 120                      }
 121                      ConflictResolution::Merged { result } => {
 122                          merged_results.push(result.clone());
 123                      }
 124                  }
 125              }
 126          }
 127          
 128          // Apply merged results after iteration
 129          for result in merged_results {
 130              self.apply_merged_result(&op, result);
 131          }
 132  
 133          // 4. Update reverse dependency index
 134          for dep_id in &op.dependencies {
 135              self.dependents
 136                  .entry(dep_id.clone())
 137                  .or_insert_with(BTreeSet::new)
 138                  .insert(op.id.clone());
 139          }
 140  
 141          // 5. Track roots
 142          if op.dependencies.is_empty() {
 143              self.roots.insert(op.id.clone());
 144          }
 145  
 146          // 6. Merge version vectors
 147          self.version_vector = self.version_vector.merge(&op.version_vector);
 148  
 149          // 7. Apply operation to state
 150          self.apply_operation(&op);
 151  
 152          // 8. Insert operation
 153          self.operations.insert(op.id.clone(), op);
 154  
 155          // 9. Update snapshot manager
 156          self.snapshot_manager.increment_operation_count();
 157  
 158          // 10. Check if we should snapshot
 159          if self.snapshot_manager.should_snapshot() {
 160              self.create_snapshot();
 161          }
 162  
 163          // 11. Garbage collect old operations
 164          self.garbage_collect();
 165  
 166          Ok(())
 167      }
