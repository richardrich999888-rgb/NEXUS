 238      pub fn calculate_savings(&self, stats: &SyncStats, total_operations: usize) -> SyncSavings {
 239          let full_replication_size = total_operations * 500;
 240          let hierarchical_size = stats.total_bytes;
 241          
 242          let bandwidth_saved = full_replication_size.saturating_sub(hierarchical_size);
 243          let savings_percent = if full_replication_size > 0 {
 244              (bandwidth_saved as f64 / full_replication_size as f64) * 100.0
 245          } else {
 246              0.0
 247          };
 248  
 249          SyncSavings {
 250              full_replication_bytes: full_replication_size,
 251              hierarchical_bytes: hierarchical_size,
 252              bytes_saved: bandwidth_saved,
 253              savings_percent,
 254          }
 255      }

 282      /// Choose sync strategy based on partition duration
 283      pub fn sync_strategy(&self, last_sync: Instant) -> SyncStrategy {
 284          if last_sync.elapsed() > self.partition_threshold {
 285              SyncStrategy::Hierarchical
 286          } else {
 287              SyncStrategy::MerkleDiff
 288          }
 289      }
