 534      #[test]
 535      fn test_rga_concurrent_inserts() {
 536          let mut doc1 = RGAText::new("node1".to_string());
 537          let mut doc2 = RGAText::new("node2".to_string());
 538  
 539          doc1.insert(0, 'A');
 540          doc2.insert(0, 'B');
 541  
 542          // Merge
 543          for char_obj in doc2.get_all_characters() {
 544              doc1.insert_remote(char_obj);
 545          }
 546          for char_obj in doc1.get_all_characters() {
 547              doc2.insert_remote(char_obj.clone());
 548          }
 549  
 550          // Both should converge
 551          assert_eq!(doc1.to_string(), doc2.to_string());
 552      }
