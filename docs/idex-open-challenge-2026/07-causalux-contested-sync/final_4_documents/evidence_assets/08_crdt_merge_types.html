 291  #[derive(Clone, Debug, Serialize, Deserialize, PartialEq, Eq, Hash, PartialOrd, Ord)]
 292  pub struct ElementId {
 293      pub value_hash: String,
 294      pub timestamp: u64,
 295      pub node_id: String,
 296  }
 297  
 298  #[derive(Clone, Debug, Serialize, Deserialize)]
 299  pub struct ORSet<T: Clone + Serialize> {
 300      elements: HashMap<String, BTreeSet<ElementId>>,
 301      removed: BTreeSet<ElementId>,
 302      node_id: String,
 303      _phantom: std::marker::PhantomData<T>,
 304  }
 305  
 306  impl<T: Clone + Serialize> ORSet<T> {
 307      pub fn new(node_id: String) -> Self {
 308          Self {
 309              elements: HashMap::new(),
 310              removed: BTreeSet::new(),
 311              node_id,
 312              _phantom: std::marker::PhantomData,
 313          }
 314      }
 315  
 316      pub fn add(&mut self, value: T) -> ElementId {
 317          let value_hash = Self::hash_value(&value);
 318          let timestamp = std::time::SystemTime::now()
 319              .duration_since(std::time::UNIX_EPOCH)
 320              .unwrap()
 321              .as_micros() as u64;
 322  
 323          let element_id = ElementId {
 324              value_hash: value_hash.clone(),
 325              timestamp,
 326              node_id: self.node_id.clone(),
 327          };
 328  
 329          self.elements
 330              .entry(value_hash)
 331              .or_insert_with(BTreeSet::new)
 332              .insert(element_id.clone());
 333  
 334          element_id
 335      }
 336  
 337      pub fn remove(&mut self, value: &T) {
 338          let value_hash = Self::hash_value(value);
 339          
 340          if let Some(element_ids) = self.elements.get(&value_hash) {
 341              for id in element_ids {
 342                  self.removed.insert(id.clone());
 343              }
 344          }
 345      }
 346  
 347      pub fn contains(&self, value: &T) -> bool {
 348          let value_hash = Self::hash_value(value);
 349          
 350          if let Some(element_ids) = self.elements.get(&value_hash) {
 351              element_ids.iter().any(|id| !self.removed.contains(id))
 352          } else {
 353              false
 354          }
 355      }
 356  
 357      pub fn elements(&self) -> Vec<String> {
 358          self.elements
 359              .iter()
 360              .filter(|(_, ids)| ids.iter().any(|id| !self.removed.contains(id)))
 361              .map(|(value_hash, _)| value_hash.clone())
 362              .collect()
 363      }
 364  
 365      pub fn merge(&mut self, other: &ORSet<T>) {
 366          for (value_hash, ids) in &other.elements {
 367              self.elements
 368                  .entry(value_hash.clone())
 369                  .or_insert_with(BTreeSet::new)
 370                  .extend(ids.clone());
 371          }
 372          self.removed.extend(other.removed.clone());
 373      }
 374  
 375      fn hash_value(value: &T) -> String {
 376          let json = serde_json::to_string(value).unwrap();
 377          let mut hasher = Sha256::new();
 378          hasher.update(json.as_bytes());
 379          format!("{:x}", hasher.finalize())
 380      }
 381  }

 387  #[derive(Clone, Debug, Serialize, Deserialize)]
 388  pub struct LWWMap<K: Clone + Eq + std::hash::Hash, V: Clone> {
 389      entries: HashMap<K, LWWRegister<Option<V>>>,
 390      node_id: String,
 391  }
 392  
 393  impl<K: Clone + Eq + std::hash::Hash, V: Clone> LWWMap<K, V> {
 394      pub fn new(node_id: String) -> Self {
 395          Self {
 396              entries: HashMap::new(),
 397              node_id,
 398          }
 399      }
 400  
 401      pub fn insert(&mut self, key: K, value: V) {
 402          let register = self.entries
 403              .entry(key)
 404              .or_insert_with(|| LWWRegister::new(None, self.node_id.clone()));
 405          register.set(Some(value));
 406      }
 407  
 408      pub fn remove(&mut self, key: &K) {
 409          if let Some(register) = self.entries.get_mut(key) {
 410              register.set(None);
 411          }
 412      }
 413  
 414      pub fn get(&self, key: &K) -> Option<&V> {
 415          self.entries.get(key).and_then(|reg| reg.get().as_ref())
 416      }
 417  
 418      pub fn iter(&self) -> impl Iterator<Item = (&K, &V)> {
 419          self.entries.iter().filter_map(|(k, reg)| {
 420              reg.get().as_ref().map(|v| (k, v))
 421          })
 422      }
 423  
 424      pub fn merge(&mut self, other: &LWWMap<K, V>) {
 425          for (key, other_register) in &other.entries {
 426              let register = self.entries
 427                  .entry(key.clone())
 428                  .or_insert_with(|| LWWRegister::new(None, self.node_id.clone()));
 429              register.merge(other_register);
 430          }
 431      }
