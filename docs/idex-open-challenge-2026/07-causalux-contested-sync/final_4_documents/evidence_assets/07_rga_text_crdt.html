  38  pub struct RGAText {

  55      /// Insert a character at a position
  56      pub fn insert(&mut self, position: usize, character: char) -> CharacterId {
  57          let timestamp = Self::current_timestamp();
  58          self.sequence_counter += 1;
  59          
  60          let id = CharacterId::new(timestamp, self.node_id.clone(), self.sequence_counter);
  61          
  62          let inserted_after = if position > 0 && !self.order.is_empty() {
  63              self.order.get(position - 1).cloned()
  64          } else {
  65              None
  66          };
  67  
  68          let char_obj = RGACharacter {
  69              id: id.clone(),
  70              value: character,
  71              deleted: false,
  72              inserted_after,
  73          };
  74  
  75          self.characters.insert(id.clone(), char_obj);
  76          let insert_pos = position.min(self.order.len());
  77          self.order.insert(insert_pos, id.clone());
  78  
  79          id
  80      }

  82      /// Delete a character at a position

  84          if position >= self.order.len() {

  98      /// Insert a character received from another node
  99      pub fn insert_remote(&mut self, char_obj: RGACharacter) {
 100          let id = char_obj.id.clone();
 101  
 102          if self.characters.contains_key(&id) {
 103              return; // Already have this character
 104          }
 105  
 106          let position = if let Some(after_id) = &char_obj.inserted_after {
 107              self.order.iter().position(|cid| cid == after_id)
 108                  .map(|pos| pos + 1)
 109                  .unwrap_or(0)
 110          } else {
 111              0
 112          };

 114          // Handle concurrent insertions - deterministic ordering
 115          let mut insert_pos = position;
 116          while insert_pos < self.order.len() {
 117              let existing_id = &self.order[insert_pos];
 118              if &id < existing_id {
 119                  break;
 120              }
 121              insert_pos += 1;
 122          }

 124          self.characters.insert(id.clone(), char_obj);
 125          self.order.insert(insert_pos, id);
