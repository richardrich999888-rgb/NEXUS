Version Vector Causality

Version vectors track happens-before ordering, conflict detection, and deterministic merge context.

  29  pub struct VersionVector {
  30      /// Map of node_id -> operation_count
  31      pub versions: BTreeMap<String, u64>,
  32  }
  33  
  34  impl VersionVector {
  35      /// Create a new empty version vector
  36      pub fn new() -> Self {
  37          Self::default()
  38      }
  39  
  40      /// Increment the count for a given node and return the new count
  41      pub fn increment(&mut self, node_id: &str) -> u64 {
  42          let counter = self.versions.entry(node_id.to_string()).or_insert(0);
  43          *counter += 1;
  44          *counter
  45      }
  46  
  47      /// Check if self happened-before other (self ≤ other)
  48      /// 
  49      /// Returns true if all entries in self are less than or equal to
  50      /// the corresponding entries in other.
  51      pub fn happens_before(&self, other: &VersionVector) -> bool {
  52          self.versions.iter().all(|(node, count)| {
  53              other.versions.get(node).map_or(false, |other_count| count <= other_count)
  54          })
  55      }
  56  
  57      /// Check if vectors conflict (neither dominates the other)
  58      /// 
  59      /// Returns true when concurrent operations from different nodes
  60      /// exist, requiring conflict resolution.
  61      pub fn conflicts_with(&self, other: &VersionVector) -> bool {
  62          !self.happens_before(other) && !other.happens_before(self)
  63      }
  64  
  65      /// Merge two version vectors (take max per node)
  66      /// 
  67      /// Creates a new version vector that represents the union of
  68      /// causality from both input vectors.
  69      pub fn merge(&self, other: &VersionVector) -> VersionVector {
  70          let mut merged = self.versions.clone();
  71          for (node, count) in &other.versions {
  72              merged
  73                  .entry(node.clone())
  74                  .and_modify(|c| *c = (*c).max(*count))
  75                  .or_insert(*count);
  76          }
  77          VersionVector { versions: merged }
  78      }
  79  
  80      /// Get the count for a specific node (0 if not present)
  81      pub fn get(&self, node_id: &str) -> u64 {
  82          self.versions.get(node_id).copied().unwrap_or(0)
  83      }
  84  
  85      /// Get the total number of operations across all nodes
  86      pub fn total_operations(&self) -> u64 {
  87          self.versions.values().sum()
  88      }
  89  
  90      /// Check if this vector is empty
  91      pub fn is_empty(&self) -> bool {
  92          self.versions.is_empty()
  93      }
