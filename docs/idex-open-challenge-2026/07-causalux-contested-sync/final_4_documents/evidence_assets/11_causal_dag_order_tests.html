 376      #[test]
 377      fn test_causal_ordering() {
 378          let keypair = create_test_keypair();
 379          let mut dag = CausalDAG::new(
 380              "node1".to_string(),
 381              100,
 382              ConflictPolicy::LastWriterWins,
 383          );
 384  
 385          let mut vv1 = VersionVector::new();
 386          vv1.increment("node1");
 387  
 388          let op1 = CausalOp::new(
 389              "op1".to_string(),
 390              serde_json::json!({}),
 391              BTreeSet::new(),
 392              vv1.clone(),
 393              "node1".to_string(),
 394              &keypair,
 395          );
 396  
 397          dag.insert(op1.clone()).unwrap();
 398  
 399          let mut vv2 = vv1.clone();
 400          vv2.increment("node1");
 401  
 402          let mut deps = BTreeSet::new();
 403          deps.insert(op1.id.clone());
 404  
 405          let op2 = CausalOp::new(
 406              "op2".to_string(),
 407              serde_json::json!({}),
 408              deps,
 409              vv2,
 410              "node1".to_string(),
 411              &keypair,
 412          );
 413  
 414          dag.insert(op2.clone()).unwrap();
 415  
 416          let order = dag.causal_order();
 417          assert_eq!(order.len(), 2);
 418          assert_eq!(order[0], op1.id);
 419          assert_eq!(order[1], op2.id);
 420      }
