  29  impl CompressedUSO {
  30      /// Create compressed USO from regular USO
  31      pub fn from_uso(uso: &USO) -> Self {
  32          let compressed_data = compress_data(&uso.data);
  33          
  34          CompressedUSO {
  35              uso_id: uso.id,
  36              compressed_data,
  37              access: uso.access.clone(),
  38              sync: uso.sync.clone(),
  39              lamport: uso.history.lamport(),
  40          }
  41      }

  43      /// Decompress to get original data
  44      pub fn decompress_data(&self) -> VectraResult<Vec<u8>> {
  45          decompress_data(&self.compressed_data)

  48      /// Get compression ratio
  49      pub fn compression_ratio(&self) -> f64 {
  50          if self.compressed_data.compressed_size == 0 {
  51              1.0
  52          } else {
  53              self.compressed_data.original_size as f64 / 
  54              self.compressed_data.compressed_size as f64
  55          }
  56      }
  57      
  58      /// Check if data was actually compressed
  59      pub fn is_compressed(&self) -> bool {
  60          self.compressed_data.was_compressed
  61      }
  62      
  63      /// Get space savings in bytes
  64      pub fn bytes_saved(&self) -> usize {
  65          if self.compressed_data.was_compressed {
  66              self.compressed_data.original_size - self.compressed_data.compressed_size
  67          } else {
  68              0
  69          }
  70      }
  71  }

  74  pub fn compress_uso_batch(usos: &[USO]) -> Vec<CompressedUSO> {

  78  /// Get total compression stats for a batch
  79  pub fn batch_compression_stats(compressed: &[CompressedUSO]) -> BatchStats {
  80      let mut stats = BatchStats::default();
  81      
  82      for uso in compressed {
  83          stats.total_count += 1;
  84          stats.original_bytes += uso.compressed_data.original_size;
  85          stats.compressed_bytes += uso.compressed_data.compressed_size;
  86          
  87          if uso.is_compressed() {
  88              stats.compressed_count += 1;
  89          }
  90      }
  91      
  92      stats
  93  }

 104  impl BatchStats {
 105      pub fn ratio(&self) -> f64 {
 106          if self.compressed_bytes == 0 {
 107              1.0
 108          } else {
 109              self.original_bytes as f64 / self.compressed_bytes as f64
 110          }
 111      }
 112      
 113      pub fn savings_percent(&self) -> f64 {
 114          if self.original_bytes == 0 {
 115              0.0
 116          } else {
 117              100.0 * (1.0 - (self.compressed_bytes as f64 / self.original_bytes as f64))
 118          }
 119      }
