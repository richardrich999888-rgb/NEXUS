  89      pub fn new(batch_size: usize, compression_enabled: bool) -> Self {
  90          Self {
  91              local_snapshots: vec![],
  92              local_operations: vec![],
  93              batch_size,

  98      /// Add a snapshot to local storage
  99      pub fn add_snapshot(&mut self, snapshot: Snapshot) {
 100          self.local_snapshots.push(snapshot);
 101      }
 102  
 103      /// Add operations to local storage
 104      pub fn add_operations(&mut self, ops: Vec<CausalOp>) {
 105          self.local_operations.extend(ops);
 106      }
 107  
 108      /// Prepare sync request
 109      pub fn prepare_sync_request(
 110          &self,
 111          node_id: String,
 112          version_vector: VersionVector,
 113          merkle_root: String,
 114      ) -> SyncRequest {
 115          SyncRequest {
 116              node_id,
 117              version_vector,
 118              latest_snapshot_id: self.local_snapshots.last().map(|s| s.id.clone()),
 119              merkle_root,
 120          }
 121      }
 122  
 123      /// Handle incoming sync request
 124      pub fn handle_sync_request(&self, request: SyncRequest) -> SyncResponse {
 125          let common_snapshot = self.find_common_snapshot(&request);
 126  
 127          if let Some(common) = common_snapshot {
 128              let ops_after = self.get_operations_after_snapshot(&common.id);
 129              
 130              SyncResponse {
 131                  common_snapshot: Some(common.clone()),
 132                  snapshot_to_download: None,
 133                  operations_after_snapshot: ops_after.clone(),
 134                  total_operations: ops_after.len(),
 135                  estimated_size_bytes: self.estimate_size(&ops_after),
 136              }
 137          } else {
 138              let latest = self.local_snapshots.last();
 139              let ops = if let Some(snap) = latest {
 140                  self.get_operations_after_snapshot(&snap.id)
 141              } else {
 142                  self.local_operations.clone()
 143              };
 144  
 145              SyncResponse {
 146                  common_snapshot: None,
 147                  snapshot_to_download: latest.map(|s| s.id.clone()),
 148                  operations_after_snapshot: ops.clone(),
 149                  total_operations: ops.len(),
 150                  estimated_size_bytes: self.estimate_snapshot_size(latest) + self.estimate_size(&ops),
 151              }
 152          }
 153      }
