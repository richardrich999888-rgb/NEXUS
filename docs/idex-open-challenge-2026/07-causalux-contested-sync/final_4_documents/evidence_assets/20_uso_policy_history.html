  81  pub enum SyncPolicy {
  82      /// Sync everywhere immediately (global consistency)
  83      Global {
  84          /// Maximum acceptable replication latency
  85          max_latency_ms: u32,
  86      },
  87      
  88      /// Sync to specific regions only

  90          regions: Vec<Region>,
  91      },
  92      
  93      /// Sync on-demand (pull-based, lazy)
  94      OnDemand,
  95      
  96      /// Local-only (single node, no sync)

  98  }

 101      fn default() -> Self {
 102          SyncPolicy::OnDemand
 103      }
 104  }

 106  impl SyncPolicy {
 107      /// Quick global sync (100ms latency target)
 108      pub fn global_fast() -> Self {
 109          SyncPolicy::Global { max_latency_ms: 100 }
 110      }
 111  
 112      /// Relaxed global sync (1s latency target)

 114          SyncPolicy::Global { max_latency_ms: 1000 }

 127  /// Access control for USO
 128  #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
 129  pub struct AccessPolicy {
 130      /// Owner (full control)
 131      pub owner: PrincipalId,
 132      
 133      /// Principals with read access
 134      pub readers: Vec<PrincipalId>,
 135      
 136      /// Principals with write access
 137      pub writers: Vec<PrincipalId>,
 138      
 139      /// Public read access
 140      pub public_read: bool,
 141      
 142      /// Public write access (dangerous!)
 143      pub public_write: bool,
 144  }
 145  
 146  impl AccessPolicy {
 147      /// Create with single owner
 148      pub fn owner_only(owner: PrincipalId) -> Self {
 149          AccessPolicy {
 150              owner,
 151              readers: Vec::new(),
 152              writers: Vec::new(),
 153              public_read: false,
 154              public_write: false,
 155          }
 156      }
 157  
 158      /// Create public read-only
 159      pub fn public_readonly(owner: PrincipalId) -> Self {
 160          AccessPolicy {
 161              owner,
 162              readers: Vec::new(),
 163              writers: Vec::new(),
 164              public_read: true,
 165              public_write: false,
 166          }
 167      }
 168  
 169      /// Check if principal can read
 170      pub fn can_read(&self, principal: &PrincipalId) -> bool {
 171          self.public_read || 
 172          *principal == self.owner ||
 173          self.readers.contains(principal) ||
 174          self.writers.contains(principal)
 175      }
 176  
 177      /// Check if principal can write
 178      pub fn can_write(&self, principal: &PrincipalId) -> bool {
 179          self.public_write ||
 180          *principal == self.owner ||
 181          self.writers.contains(principal)
 182      }
 183  }

 225  /// Causal history for USO (enables CRDT-style merge)
 226  #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
 227  pub struct CausalHistory {
 228      /// Vector clock for causality tracking
 229      pub vector_clock: HashMap<String, u64>,
 230      
 231      /// Operation log
 232      pub operations: Vec<HistoryEntry>,
 233      
 234      /// Parent states this was derived from
 235      pub parents: Vec<ContentHash>,
 236  }
 237  
 238  impl CausalHistory {
 239      /// Create new empty history
 240      pub fn new() -> Self {
 241          CausalHistory {
 242              vector_clock: HashMap::new(),
 243              operations: Vec::new(),
 244              parents: Vec::new(),
 245          }
 246      }
 247  
 248      /// Create genesis history
 249      pub fn genesis() -> Self {
 250          CausalHistory::new()
 251      }
 252  
 253      /// Tick the vector clock for a node
 254      pub fn tick(&mut self, node_id: &str) -> u64 {
 255          let counter = self.vector_clock.entry(node_id.to_string()).or_insert(0);
 256          *counter += 1;
 257          *counter
 258      }
 259  
 260      /// Get current Lamport timestamp
 261      pub fn lamport(&self) -> u64 {
 262          self.vector_clock.values().copied().max().unwrap_or(0)
 263      }
 264  
 265      /// Add operation to history
 266      pub fn add_operation(&mut self, operation: Operation, principal: PrincipalId, node_id: &str) {
 267          let lamport = self.tick(node_id);
 268          self.operations.push(HistoryEntry {
 269              operation,
 270              principal,
 271              lamport,
 272              timestamp: crate::now(),
 273          });
 274      }
 275  
 276      /// Merge with another history (CRDT-style)
 277      pub fn merge(&mut self, other: &CausalHistory) {
 278          // Merge vector clocks (take max)
 279          for (node, &counter) in &other.vector_clock {
 280              let entry = self.vector_clock.entry(node.clone()).or_insert(0);
 281              *entry = (*entry).max(counter);
 282          }
 283  
 284          // Merge operations (by Lamport + dedup)
 285          for op in &other.operations {
 286              if !self.operations.iter().any(|o| o.lamport == op.lamport && o.principal == op.principal) {
 287                  self.operations.push(op.clone());
 288              }
 289          }
 290  
 291          // Sort by Lamport for deterministic order
 292          self.operations.sort_by_key(|o| o.lamport);
 293      }
 294  
 295      /// Check if this history happens-before another
 296      pub fn happens_before(&self, other: &CausalHistory) -> bool {
 297          // Self happens before other if all our clocks <= other's and at least one <
 298          let mut at_least_one_less = false;
 299          
 300          for (node, &counter) in &self.vector_clock {
 301              let other_counter = other.vector_clock.get(node).copied().unwrap_or(0);
 302              if counter > other_counter {
 303                  return false;
 304              }
 305              if counter < other_counter {
 306                  at_least_one_less = true;
 307              }
 308          }
 309          
 310          at_least_one_less
 311      }
 312  }
