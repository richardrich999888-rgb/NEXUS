  47  /// Compression statistics
  48  #[derive(Clone, Debug, Default, Serialize, Deserialize)]
  49  pub struct CompressionStats {
  50      /// Total original bytes
  51      pub original_bytes: usize,
  52      
  53      /// Total compressed bytes
  54      pub compressed_bytes: usize,
  55      
  56      /// Number of inputs compressed
  57      pub inputs_compressed: usize,
  58      
  59      /// Number of inputs passed through
  60      pub inputs_passthrough: usize,
  61  }

  63  impl CompressionStats {
  64      /// Calculate compression ratio
  65      pub fn ratio(&self) -> f64 {
  66          if self.compressed_bytes == 0 {
  67              1.0
  68          } else {
  69              self.original_bytes as f64 / self.compressed_bytes as f64
  70          }
  71      }
  72      
  73      /// Calculate space savings percentage
  74      pub fn savings_percent(&self) -> f64 {
  75          if self.original_bytes == 0 {
  76              0.0
  77          } else {
  78              100.0 * (1.0 - (self.compressed_bytes as f64 / self.original_bytes as f64))
  79          }
  80      }
  81  }

  83  /// Compress a single data block
  84  pub fn compress_data(data: &[u8]) -> CompressedData {
  85      let original_hash = ContentHash::compute(data);
  86      let original_size = data.len();
  87      
  88      // Create VECTRA Payload from bytes
  89      let payload = Payload::new(data.to_vec());
  90      
  91      match vectra_encode(payload) {
  92          EncodeResult::Encoded(artifact) => {
  93              let artifact_bytes = artifact.to_bytes();
  94              let compressed_size = artifact_bytes.len();
  95              
  96              CompressedData {
  97                  original_hash,
  98                  original_size,
  99                  artifact_bytes,
 100                  compressed_size,
 101                  was_compressed: true,
 102              }
 103          }
 104          EncodeResult::PassThrough(original_payload) => {
 105              // PassThrough returns the original Payload, extract bytes
 106              let original_bytes = original_payload.as_bytes().to_vec();
 107              CompressedData {
 108                  original_hash,
 109                  original_size,
 110                  artifact_bytes: original_bytes,
 111                  compressed_size: original_size,
 112                  was_compressed: false,
 113              }
 114          }
 115      }
 116  }

 118  /// Decompress a data block
 119  pub fn decompress_data(compressed: &CompressedData) -> VectraResult<Vec<u8>> {
 120      if !compressed.was_compressed {
 121          return Ok(compressed.artifact_bytes.clone());
 122      }
 123      
 124      let artifact = Artifact::from_bytes(&compressed.artifact_bytes)?;
 125      let payload = vectra_decode(&artifact)?;
 126      Ok(payload.as_bytes().to_vec())
 127  }

 129  /// Compress PCU inputs and parameters
 130  pub fn compress_pcu(pcu: &PCU, input_data: &[Vec<u8>]) -> CompressedPCU {
 131      let mut stats = CompressionStats::default();
 132      let mut compressed_inputs = Vec::new();
 133      
 134      for data in input_data {
 135          let compressed = compress_data(data);
 136          
 137          stats.original_bytes += compressed.original_size;
 138          stats.compressed_bytes += compressed.compressed_size;
 139          
 140          if compressed.was_compressed {
 141              stats.inputs_compressed += 1;
 142          } else {
 143              stats.inputs_passthrough += 1;
 144          }
 145          
 146          compressed_inputs.push(compressed);
 147      }
 148      
 149      // Compress parameters if present
 150      let compressed_params = if !pcu.parameters.is_empty() {
 151          let params = compress_data(&pcu.parameters);
 152          stats.original_bytes += params.original_size;
 153          stats.compressed_bytes += params.compressed_size;
 154          Some(params)
 155      } else {
 156          None
 157      };
 158      
 159      CompressedPCU {
 160          pcu_id: pcu.id,
 161          compressed_inputs,
 162          compressed_params,
 163          code_hash: pcu.code.hash,
 164          stats,
 165      }
 166  }

 168  /// Decompress all PCU data
 169  pub fn decompress_pcu(compressed: &CompressedPCU) -> VectraResult<Vec<Vec<u8>>> {
 170      let mut inputs = Vec::new();
 171      
 172      for comp_data in &compressed.compressed_inputs {
 173          inputs.push(decompress_data(comp_data)?);
 174      }
 175      
 176      Ok(inputs)
 177  }
