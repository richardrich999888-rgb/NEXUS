 247      /// Merge with another CRDT USO of the same type
 248      pub fn merge(&mut self, other: &CrdtUSO) -> Result<(), CrdtError> {
 249          match (&mut self.state, &other.state) {
 250              (CrdtState::Counter(a), CrdtState::Counter(b)) => {
 251                  a.merge(b);
 252                  self.update_id();
 253                  Ok(())
 254              }
 255              (CrdtState::PNCounter(a), CrdtState::PNCounter(b)) => {
 256                  a.merge(b);
 257                  self.update_id();
 258                  Ok(())
 259              }
 260              (CrdtState::Set(a), CrdtState::Set(b)) => {
 261                  a.merge(b);
 262                  self.update_id();
 263                  Ok(())
 264              }
 265              _ => Err(CrdtError::TypeMismatch),
 266          }
 267      }

 269      /// Update content-addressed ID after mutation
 270      fn update_id(&mut self) {
 271          let bytes = match &self.state {
 272              CrdtState::Raw(data) => data.clone(),
 273              CrdtState::Json(v) => v.to_string().into_bytes(),
 274              CrdtState::Counter(c) => c.value().to_le_bytes().to_vec(),
 275              CrdtState::PNCounter(c) => c.value().to_le_bytes().to_vec(),
 276              CrdtState::Register(r) => serde_json::to_vec(r).unwrap_or_default(),
 277              CrdtState::Set(s) => serde_json::to_vec(s).unwrap_or_default(),
 278              CrdtState::Map(m) => serde_json::to_vec(m).unwrap_or_default(),
 279              CrdtState::Text(t) => t.to_string().into_bytes(),
 280          };
 281          self.id = ContentHash::compute(&bytes);
 282      }
