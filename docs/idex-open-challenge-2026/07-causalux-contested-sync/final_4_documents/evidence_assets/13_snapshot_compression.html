Snapshots and Compression

Snapshots carry Merkle root, version vector, operation count, compressed size, and common-snapshot negotiation.
  13  /// A point-in-time snapshot of the system state.
  14  /// 
  15  /// Snapshots enable:
  16  /// - Constant memory footprint (via garbage collection)
  17  /// - Fast sync after long partitions (download snapshot, not full history)
  18  /// - Efficient state reconstruction
  19  #[derive(Clone, Debug, Serialize, Deserialize)]
  20  pub struct Snapshot {
  21      /// Unique snapshot ID
  22      pub id: String,
  23      
  24      /// Complete state at this point (serialized)
  25      pub state: serde_json::Value,
  26      
  27      /// Merkle root of all operations up to this snapshot
  28      pub merkle_root: String,
  29      
  30      /// Version vector at snapshot time
  31      pub version_vector: VersionVector,
  32      
  33      /// Unix timestamp when created
  34      pub timestamp: u64,
  35      
  36      /// Number of operations included
  37      pub operation_count: u64,
  38      
  39      /// Compressed size in bytes
  40      pub compressed_size: usize,
  41  }
  42  
  43  impl Snapshot {
  44      /// Create a new snapshot
  45      pub fn new(
  46          state: serde_json::Value,
  47          merkle_root: String,
  48          version_vector: VersionVector,
  49          operation_count: u64,
  50      ) -> Self {
  51          let serialized = serde_json::to_string(&state).unwrap();
  52          let compressed = Self::compress_state(&serialized);
  53          let compressed_size = compressed.len();
  54  
  55          // Generate unique ID from content hash
  56          let id = Self::generate_id(&merkle_root, operation_count);
  57  
  58          Self {
  59              id,
  60              state,
  61              merkle_root,
  62              version_vector,
  63              timestamp: SystemTime::now()
  64                  .duration_since(UNIX_EPOCH)
  65                  .unwrap()
  66                  .as_secs(),
  67              operation_count,
  68              compressed_size,
  69          }
  70      }
  71  
  72      /// Compress state for storage/transfer
  73      pub fn compress_state(state: &str) -> Vec<u8> {
  74          let mut encoder = GzEncoder::new(Vec::new(), Compression::best());
  75          encoder.write_all(state.as_bytes()).unwrap();
  76          encoder.finish().unwrap()
  77      }
  78  
  79      /// Decompress state
  80      pub fn decompress_state(compressed: &[u8]) -> String {
  81          let mut decoder = GzDecoder::new(compressed);
  82          let mut decompressed = String::new();
  83          decoder.read_to_string(&mut decompressed).unwrap();
  84          decompressed
  85      }
  86  
  87      /// Get compressed bytes for transfer
  88      pub fn to_compressed_bytes(&self) -> Vec<u8> {
  89          let json = serde_json::to_string(&self.state).unwrap();
  90          Self::compress_state(&json)
  91      }
  92  
  93      /// Generate snapshot ID from content
  94      fn generate_id(merkle_root: &str, operation_count: u64) -> String {
  95          let mut hasher = Sha256::new();
  96          hasher.update(merkle_root.as_bytes());
  97          hasher.update(operation_count.to_le_bytes());
  98          format!("snap_{:x}", hasher.finalize())[..24].to_string()
  99      }
 100  }

 206      /// Find most recent common snapshot with a peer
 207      pub fn find_common_snapshot(&self, peer_snapshot_ids: &[String]) -> Option<&Snapshot> {
 208          // Iterate from most recent to oldest
 209          for snapshot in self.snapshots.iter().rev() {
 210              if peer_snapshot_ids.contains(&snapshot.id) {
 211                  return Some(snapshot);
 212              }
 213          }
 214          None
 215      }
 216  
 217      /// Get memory footprint estimate
 218      pub fn memory_footprint(&self) -> usize {
 219          self.snapshots.iter().map(|s| s.compressed_size).sum()
 220      }
 221  
 222      /// Get total operations count
 223      pub fn total_operations(&self) -> u64 {
 224          self.total_operations
 225      }
 226  
 227      /// Get snapshot count
 228      pub fn snapshot_count(&self) -> usize {
 229          self.snapshots.len()
 230      }
 231  }
