NEXUS Sync Engine

The sync engine wraps CAUSALUX DAG with USO registry, signed operations, sync deltas, and remote merge.
  11  /// NEXUS Sync Engine - manages synchronized state across nodes
  12  /// 
  13  /// Wraps CAUSALUX CausalDAG with NEXUS-specific adaptations:
  14  /// - Uses ContentHash for operation IDs
  15  /// - Integrates with USO for state management
  16  /// - Provides PCU-aware sync protocols
  17  pub struct NexusSyncEngine {
  18      /// Underlying CAUSALUX DAG
  19      dag: CausalDAG,
  20      
  21      /// Node ID
  22      node_id: String,
  23      
  24      /// Keypair for signing operations
  25      keypair: Option<SigningKey>,
  26      
  27      /// USO registry - tracked USOs by their ID
  28      uso_registry: HashMap<ContentHash, USO>,
  29  }
  30  
  31  impl NexusSyncEngine {
  32      /// Create new sync engine
  33      pub fn new(node_id: impl Into<String>, conflict_policy: ConflictPolicy) -> Self {
  34          let node_id = node_id.into();
  35          let dag = CausalDAG::new(
  36              node_id.clone(),
  37              10_000, // Snapshot every 10K ops
  38              conflict_policy,
  39          );
  40          
  41          NexusSyncEngine {
  42              dag,
  43              node_id,
  44              keypair: None,
  45              uso_registry: HashMap::new(),
  46          }
  47      }
  48  
  49      /// Set keypair for signing operations
  50      pub fn with_keypair(mut self, keypair: SigningKey) -> Self {
  51          self.keypair = Some(keypair);
  52          self
  53      }
  54  
  55      /// Get current version vector
  56      pub fn version_vector(&self) -> &VersionVector {
  57          self.dag.get_version_vector()
  58      }
  59  
  60      /// Get operation count
  61      pub fn operation_count(&self) -> usize {
  62          self.dag.operation_count()
  63      }
  64  
  65      /// Register a USO for sync tracking
  66      pub fn register_uso(&mut self, uso: USO) {
  67          self.uso_registry.insert(uso.id, uso);
  68      }
  69  
  70      /// Get a tracked USO
  71      pub fn get_uso(&self, id: &ContentHash) -> Option<&USO> {
  72          self.uso_registry.get(id)
  73      }
  74  
  75      /// Update a USO and create sync operation
  76      pub fn update_uso(
  77          &mut self,
  78          uso_id: &ContentHash,
  79          new_data: Vec<u8>,
  80          principal: PrincipalId,
  81      ) -> Result<ContentHash, SyncError> {
  82          let uso = self.uso_registry.get_mut(uso_id)
  83              .ok_or(SyncError::UsoNotFound)?;
  84          
  85          uso.update(new_data.clone(), principal, &self.node_id);
  86          
  87          // Create causal operation for sync
  88          if let Some(ref keypair) = self.keypair {
  89              let version = uso.history.vector_clock.get(&self.node_id).copied().unwrap_or(0);
  90              let mut vv = VersionVector::new();
  91              for _ in 0..version {
  92                  vv.increment(&self.node_id);
  93              }
  94              
  95              let op = CausalOp::new(
  96                  "uso_update".to_string(),
  97                  serde_json::json!({
  98                      "uso_id": hex::encode(uso_id.as_bytes()),
  99                      "data_hash": hex::encode(uso.id.as_bytes()),
 100                  }),
 101                  std::collections::BTreeSet::new(),
 102                  vv,
 103                  self.node_id.clone(),
 104                  keypair,
 105              );
 106              
 107              self.dag.insert(op).map_err(|e| SyncError::DagError(e.to_string()))?;
 108          }
 109          
 110          Ok(uso.id)
 111      }
 112  
 113      /// Get operations since a version vector (for sync)
 114      pub fn get_operations_since(&self, since_lamport: u64) -> Vec<&CausalOp> {
 115          self.dag.get_operations_after(since_lamport)
 116      }
 117  
 118      /// Merge remote operations into local DAG
 119      pub fn merge_remote(&mut self, ops: Vec<CausalOp>) -> Result<usize, SyncError> {
 120          let mut merged = 0;
 121          for op in ops {
 122              match self.dag.insert(op) {
 123                  Ok(_) => merged += 1,
 124                  Err(e) => {
 125                      // Log but continue - some ops may conflict
 126                      eprintln!("Merge warning: {}", e);
 127                  }
 128              }
 129          }
 130          Ok(merged)
 131      }
 132  
 133      /// Get sync delta for a peer
 134      pub fn get_sync_delta(&self, peer_vv: &VersionVector) -> SyncDelta {
 135          let local_vv = self.version_vector();
 136          
 137          // Find common point
 138          let mut min_lamport = u64::MAX;
 139          for (node, count) in &peer_vv.versions {
 140              let local_count = local_vv.get(node);
 141              if local_count > *count {
 142                  min_lamport = min_lamport.min(*count);
 143              }
 144          }
 145          
 146          let ops = if min_lamport == u64::MAX {
 147              vec![]
 148          } else {
 149              self.get_operations_since(min_lamport)
 150                  .into_iter()
 151                  .cloned()
 152                  .collect()
 153          };
 154          
 155          SyncDelta {
 156              from_version: peer_vv.clone(),
 157              to_version: local_vv.clone(),
 158              operations: ops,
 159          }
 160      }
