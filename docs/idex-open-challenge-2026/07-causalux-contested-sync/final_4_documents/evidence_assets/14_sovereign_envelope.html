 212      /// Encrypt an operation into an envelope using AES-256-GCM
 213      /// 
 214      /// PRODUCTION-GRADE: Uses authenticated encryption with AES-GCM
 215      pub fn seal(op: &CausalOp, key: &DerivedKey, document_id: String) -> Result<Self, EnvelopeError> {
 216          let plaintext = serde_json::to_vec(op)
 217              .map_err(|_| EnvelopeError::InvalidContent)?;
 218          
 219          // Initialize AES-256-GCM cipher
 220          let cipher = Aes256Gcm::new_from_slice(&key.key_bytes)
 221              .map_err(|_| EnvelopeError::InvalidKey)?;
 222          
 223          // Generate cryptographically secure nonce (96 bits for GCM)
 224          let nonce_bytes = Self::generate_nonce();
 225          let nonce = Nonce::from_slice(&nonce_bytes);
 226          
 227          // Encrypt with authenticated encryption (includes authentication tag)
 228          let ciphertext_with_tag = cipher.encrypt(nonce, plaintext.as_ref())
 229              .map_err(|_| EnvelopeError::EncryptionFailure)?;
 230          
 231          // GCM appends 16-byte tag to ciphertext
 232          let (ciphertext, tag_bytes) = ciphertext_with_tag.split_at(ciphertext_with_tag.len() - 16);
 233          let mut tag = [0u8; 16];
 234          tag.copy_from_slice(tag_bytes);
 235          
 236          Ok(Self {
 237              version: 1,
 238              document_id,
 239              key_id: key.key_id.clone(),
 240              ciphertext: ciphertext.to_vec(),
 241              nonce: nonce_bytes,
 242              tag,
 243              metadata: EnvelopeMetadata {
 244                  operation_id: op.id.clone(),
 245                  creator_id: op.identity.clone(),
 246                  timestamp: op.wall_clock,
 247                  dependencies: op.dependencies.iter().cloned().collect(),
 248                  version_summary: op.version_vector.versions.iter().map(|(k, v)| (k.clone(), *v)).collect(),
 249              },
 250          })
 251      }

 253      /// Decrypt an envelope to get the operation using AES-256-GCM
 254      pub fn unseal(&self, key: &DerivedKey) -> Result<CausalOp, EnvelopeError> {
 255          // Verify key ID matches
 256          if self.key_id != key.key_id {
 257              return Err(EnvelopeError::WrongKey);
 258          }
 259          
 260          // Initialize AES-256-GCM cipher
 261          let cipher = Aes256Gcm::new_from_slice(&key.key_bytes)
 262              .map_err(|_| EnvelopeError::InvalidKey)?;
 263          
 264          let nonce = Nonce::from_slice(&self.nonce);
 265          
 266          // Reconstruct ciphertext with tag (GCM expects them together)
 267          let mut ciphertext_with_tag = self.ciphertext.clone();
 268          ciphertext_with_tag.extend_from_slice(&self.tag);
 269          
 270          // Decrypt with authentication verification
 271          let plaintext = cipher.decrypt(nonce, ciphertext_with_tag.as_ref())
 272              .map_err(|_| EnvelopeError::IntegrityFailure)?;  // Auth tag verification happens here
 273          
 274          let op: CausalOp = serde_json::from_slice(&plaintext)
 275              .map_err(|_| EnvelopeError::InvalidContent)?;

 277          Ok(op)
