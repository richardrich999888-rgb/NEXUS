 293          severity: ThreatSeverity,
 294          context: Dict[str, Any]
 295      ) -> str:
 296          """Take appropriate governance action."""
 297          level_map = {
 298              ThreatSeverity.BENIGN: ThreatLevel.NONE,
 299              ThreatSeverity.LOW: ThreatLevel.LOW,
 300              ThreatSeverity.MEDIUM: ThreatLevel.MEDIUM,
 301              ThreatSeverity.HIGH: ThreatLevel.HIGH,
 302              ThreatSeverity.CRITICAL: ThreatLevel.CRITICAL,
 303          }
 304          signal = ThreatSignal(
 305              agent_id=agent_id,
 306              threat_level=level_map[severity],
 307              threat_type=context.get("threat_type", severity.name.lower()),
 308              confidence=severity.value / ThreatSeverity.CRITICAL.value,
 309              details=context,
 310          )
 311          action = self.governance_bridge.register_threat(signal)
 312          action_name = action.get("action", "monitor")
 313  
 314          if action_name == "quarantine":
 315              self.quarantines_issued += 1
 316              return "quarantine"
 317          elif action_name == "block":
 318              return "restrict"
 319          elif action_name == "throttle":
 320              return "escalate"
 321          elif action_name == "monitor":
 322              return "warn"
 323          return action_name
