 109      def _determine_action(self, signal: ThreatSignal) -> Dict:
 110          """Determine governance action based on threat"""
 111          if signal.threat_level == ThreatLevel.NONE:
 112              return {"action": "none", "agent_id": signal.agent_id}
 113  
 114          if signal.threat_level == ThreatLevel.LOW:
 115              return {
 116                  "action": "monitor",
 117                  "agent_id": signal.agent_id,
 118                  "increase_logging": True
 119              }
 120  
 121          if signal.threat_level == ThreatLevel.MEDIUM:
 122              return {
 123                  "action": "throttle",
 124                  "agent_id": signal.agent_id,
 125                  "rate_limit": 0.5,  # 50% reduction
 126                  "duration_seconds": 300
 127              }
 128  
 129          if signal.threat_level == ThreatLevel.HIGH:
 130              return {
 131                  "action": "block",
 132                  "agent_id": signal.agent_id,
 133                  "block_actuators": True,
 134                  "escalate_to_human": True
 135              }
 136  
 137          if signal.threat_level == ThreatLevel.CRITICAL:
 138              return {
 139                  "action": "quarantine",
 140                  "agent_id": signal.agent_id,
 141                  "disconnect_mesh": True,
 142                  "freeze_resources": True,
 143                  "escalate_to_human": True,
 144                  "priority": "CRITICAL"
 145              }
