 211      def propagate_trust(self, from_agent: str, to_agent: str, weight: float = 0.1):
 212          """Propagate trust between agents in mesh"""
 213          from_trust = self.get_trust(from_agent)
 214          to_trust = self.get_trust(to_agent)
 215  
 216          # Weighted average towards from_agent's trust
 217          new_trust = to_trust + (from_trust - to_trust) * weight
 218          self.trust_scores[to_agent] = new_trust
 219  
 220          return {"agent": to_agent, "new_trust": new_trust}
