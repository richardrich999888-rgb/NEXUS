 117  #[derive(Clone, PartialEq, Eq)]
 118  pub struct ExecutionEventV1 {
 119      pub event_id: Hash256,
 120      pub execution_id: Hash256,
 121      pub sequence_number: u64,
 122      pub timestamp_utc: u64,
 123      pub actor_id: Hash256,
 124      pub workload_id: Hash256,
 125      pub execution_context: Hash256,
 126      pub resource_class: ResourceClass,
 127      pub jurisdiction_code: u16,
 128      pub policy_ref: Hash256,
 129      pub outcome_code: OutcomeCode,
 130      pub previous_event_hash: Hash256,
 131  }

 134      /// Canonical serialization: field order exactly as schema. Big-endian. No optional fields.
 135      /// Used for storage/wire. event_id is included.
 136      pub fn to_canonical_bytes(&self) -> Vec<u8> {
 137          let mut out = Vec::with_capacity(32 * 8 + 8 + 8 + 1 + 2 + 1);
 138          out.extend_from_slice(&self.event_id.0);
 139          out.extend_from_slice(&self.execution_id.0);
 140          out.extend_from_slice(&self.sequence_number.to_be_bytes());
 141          out.extend_from_slice(&self.timestamp_utc.to_be_bytes());
 142          out.extend_from_slice(&self.actor_id.0);
 143          out.extend_from_slice(&self.workload_id.0);
 144          out.extend_from_slice(&self.execution_context.0);
 145          out.push(self.resource_class.to_u8());
 146          out.extend_from_slice(&self.jurisdiction_code.to_be_bytes());
 147          out.extend_from_slice(&self.policy_ref.0);
 148          out.push(self.outcome_code.to_u8());
 149          out.extend_from_slice(&self.previous_event_hash.0);
 150          out
 151      }
