  60  /// Verify ExecutionProof and event stream. Offline; no API calls.
  61  /// Events may be in any order; verifier sorts by sequence_number and validates chain.
  62  pub fn verify(
  63      proof: &ExecutionProofV1,
  64      events: &[ExecutionEventV1],
  65      policy_resolver: &PolicyResolver,
  66      verifier_pubkey: &VerifyingKey,
  67      tolerance_ms: u64,
  68  ) -> Result<Verdict, VerifierError> {
  69      // Phase 1: Schema & version
  70      if events.is_empty() {
  71          return Err(VerifierError::E6ExecutionIncomplete);
  72      }
  73  
  74      let mut events_sorted: Vec<_> = events.to_vec();
  75      events_sorted.sort_by_key(|e| e.sequence_number);
  76  
  77      for ev in &events_sorted {
  78          if ev.compute_event_id() != ev.event_id {
  79              return Err(VerifierError::E1SchemaViolation(SchemaError::EventIdMismatch));
  80          }
  81      }
  82  
  83      // Phase 2: Signature
  84      let signing_bytes = proof.to_signing_bytes();
  85      let sig = Signature::from_bytes(&proof.verifier_signature);
  86      verifier_pubkey
  87          .verify(&signing_bytes, &sig)
  88          .map_err(|_| VerifierError::E2SignatureInvalid)?;
  89  
  90      // Phase 3: Execution consistency
  91      let execution_id = proof.execution_id;
  92      let policy_ref = proof.policy_ref;
  93      let jurisdiction_code = proof.jurisdiction_code;
  94  
  95      for (i, ev) in events_sorted.iter().enumerate() {
  96          if ev.execution_id != execution_id {
  97              return Err(VerifierError::E3HashChainBroken);
  98          }
  99          if ev.policy_ref != policy_ref {
 100              return Err(VerifierError::E3HashChainBroken);
 101          }
 102          if ev.jurisdiction_code != jurisdiction_code {
 103              return Err(VerifierError::E3HashChainBroken);
 104          }
 105          if ev.sequence_number != i as u64 {
 106              return Err(VerifierError::E3HashChainBroken);
 107          }
 108      }

 112          if i == 0 {

 114                  return Err(VerifierError::E3HashChainBroken);
 115              }
 116          } else {

 118              if ev.previous_event_hash != prev_id {
 119                  return Err(VerifierError::E3HashChainBroken);
