  83  // EXECUTION PROOF - Cryptographic proof of correct execution
  84  // ============================================================================
  85  
  86  /// Execution proof enables trust without re-verification
  87  /// 
  88  /// Contains:
  89  /// - What inputs were used (by hash)
  90  /// - What code executed (by hash)
  91  /// - What output was produced (by hash)
  92  /// - Attestation from executing node
  93  ///
  94  /// Any node can verify this proof and trust the result.
  95  #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
  96  pub struct ExecutionProof {
  97      /// Hash of the PCU that was executed
  98      pub pcu_hash: ContentHash,
  99      
 100      /// Hashes of inputs that were used
 101      pub input_hashes: Vec<ContentHash>,
 102      
 103      /// Hash of code that executed
 104      pub code_hash: ContentHash,
 105      
 106      /// Hash of output produced
 107      pub output_hash: ContentHash,
 108      
 109      /// Execution duration in microseconds (for audit)
 110      pub duration_us: u64,
 111      
 112      /// Memory used in bytes (for audit)
 113      pub memory_bytes: u64,
 114      
 115      /// Attestation from executing node
 116      pub attestation: NodeAttestation,
 117  }
 118

 120      /// Create new execution proof

 122          pcu_hash: ContentHash,

 139      }
