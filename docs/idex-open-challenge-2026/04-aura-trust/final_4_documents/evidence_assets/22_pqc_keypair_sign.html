 184  /// Default builds generate only classical keys. Builds with `--features pqc`
 185  /// generate both Ed25519 and ML-DSA-65 keys.
 186  pub struct HybridKeyPair {
 187      /// Classical Ed25519 signing key
 188      pub classical: SigningKey,
 189  
 190      /// ML-DSA-65 private key for PQC signatures
 191      #[cfg(feature = "pqc")]
 192      pub pqc_sk: Option<ml_dsa_65::PrivateKey>,
 193  
 194      /// ML-DSA-65 public key for verification
 195      #[cfg(feature = "pqc")]
 196      pub pqc_pk: Option<ml_dsa_65::PublicKey>,
 197      
 198      /// Reserved for when pqc feature is disabled
 199      #[cfg(not(feature = "pqc"))]
 200      _pqc_reserved: Option<()>,
 201  }
 202  
 203  impl HybridKeyPair {
 204      /// Generate a new keypair.
 205      pub fn generate() -> PqcResult<Self> {
 206          use rand::RngCore;
 207  
 208          // 1. Generate classical Ed25519 key
 209          let mut secret = [0u8; 32];
 210          rand::rngs::OsRng.fill_bytes(&mut secret);
 211          let classical = SigningKey::from_bytes(&secret);
 212  
 213          // 2. Generate PQC ML-DSA-65 key if enabled
 214          #[cfg(feature = "pqc")]
 215          {
 216              let mut rng = rand::rngs::OsRng;
 217              let (pk, sk) = ml_dsa_65::try_keygen_with_rng(&mut rng)
 218                  .map_err(|e| PqcError::KeyGenerationFailed(format!("{:?}", e)))?;
 219              
 220              Ok(Self {
 221                  classical,
 222                  pqc_sk: Some(sk),
 223                  pqc_pk: Some(pk),
 224              })
 225          }
 226  
 227          #[cfg(not(feature = "pqc"))]
 228          Ok(Self {
 229              classical,
 230              _pqc_reserved: None,
 231          })
 232      }
 233  
 234      /// Create from existing Ed25519 key (classical-only mode)
 235      pub fn from_classical(signing_key: SigningKey) -> Self {
 236          Self {
 237              classical: signing_key,
 238              #[cfg(feature = "pqc")]
 239              pqc_sk: None,
 240              #[cfg(feature = "pqc")]
 241              pqc_pk: None,
 242              #[cfg(not(feature = "pqc"))]
 243              _pqc_reserved: None,
 244          }
 245      }
 246  
 247      /// Get the classical verifying (public) key
 248      pub fn classical_verifying_key(&self) -> VerifyingKey {
 249          self.classical.verifying_key()
 250      }
 251  
 252      pub fn pqc_public_key(&self) -> Option<Vec<u8>> {
 253          #[cfg(feature = "pqc")]
 254          {
 255              if let Some(pk) = &self.pqc_pk {
 256                  return Some(pk.clone().into_bytes().to_vec());
 257              }
 258          }
 259          None
 260      }
 261  
 262      /// Sign a message.
 263      pub fn sign(&self, message: &[u8]) -> HybridSignature {
 264          let classical_sig = self.classical.sign(message);
 265          
 266          #[cfg(feature = "pqc")]
 267          {
 268              if let Some(sk) = &self.pqc_sk {
 269                  if let Ok(pqc_sig) = sk.try_sign(message, &[]) {
 270                      return HybridSignature::hybrid(&classical_sig, &pqc_sig);
 271                  }
 272              }
 273          }
 274          
 275          HybridSignature::classical_only(&classical_sig)
 276      }
 277  
 278      /// Check if this keypair has PQC capability
 279      pub fn has_pqc(&self) -> bool {
 280          #[cfg(feature = "pqc")]
 281          return self.pqc_sk.is_some();
 282          #[cfg(not(feature = "pqc"))]
 283          false
 284      }
 285  }

 291  /// Bundle of public keys for verification (supports both classical and PQC)
 292  #[derive(Clone, Debug, Serialize, Deserialize)]
 293  pub struct PublicKeyBundle {
 294      /// Classical Ed25519 public key (32 bytes)
 295      pub classical: Vec<u8>,
 296  
 297      /// Post-quantum ML-DSA public key (~1,952 bytes for ML-DSA-65)
 298      #[serde(skip_serializing_if = "Option::is_none")]
 299      pub pqc: Option<Vec<u8>>,
 300  
 301      /// Key bundle version
 302      pub version: u8,
 303  }
 304  
 305  impl PublicKeyBundle {
 306      /// Current key bundle version
 307      pub const VERSION: u8 = 1;
 308  
 309      /// Create classical-only bundle
 310      pub fn classical_only(public_key: &VerifyingKey) -> Self {
 311          Self {
 312              classical: public_key.to_bytes().to_vec(),
 313              pqc: None,
 314              version: Self::VERSION,
 315          }
 316      }
 317  
 318      /// Create hybrid bundle.
 319      pub fn hybrid(classical: &VerifyingKey, pqc: Vec<u8>) -> Self {
 320          Self {
 321              classical: classical.to_bytes().to_vec(),
 322              pqc: Some(pqc),
 323              version: Self::VERSION,
 324          }
 325      }
 326  
 327      /// Get Ed25519 verifying key
 328      pub fn classical_verifying_key(&self) -> PqcResult<VerifyingKey> {
 329          let bytes: [u8; 32] = self
 330              .classical
 331              .as_slice()
 332              .try_into()
 333              .map_err(|_| PqcError::InvalidKeyFormat("Invalid classical key length".into()))?;
 334  
 335          VerifyingKey::from_bytes(&bytes)
 336              .map_err(|e| PqcError::InvalidKeyFormat(e.to_string()))
 337      }
 338  
 339      /// Verify a hybrid signature
 340      pub fn verify(&self, message: &[u8], signature: &HybridSignature) -> bool {
 341          let Ok(classical_pk) = self.classical_verifying_key() else {
 342              return false;
 343          };
 344  
 345          signature.verify_hybrid(message, &classical_pk, self.pqc.as_deref())
 346      }
 347  
 348      /// Total size in bytes
 349      pub fn size(&self) -> usize {
 350          self.classical.len() + self.pqc.as_ref().map(|p| p.len()).unwrap_or(0)
 351      }
