 117      /// Verify the classical (Ed25519) signature
 118      pub fn verify_classical(&self, message: &[u8], public_key: &VerifyingKey) -> bool {
 119          if self.classical.len() != 64 {
 120              return false;
 121          }
 122  
 123          let sig_bytes: [u8; 64] = match self.classical.as_slice().try_into() {
 124              Ok(b) => b,
 125              Err(_) => return false,
 126          };
 127  
 128          let signature = Ed25519Signature::from_bytes(&sig_bytes);
 129          public_key.verify(message, &signature).is_ok()
 130      }
