Delegation links check expiry, continuity, signatures, and canonical signing data.
 222      /// Check if entire chain is valid (not expired, signatures valid)
 223      pub fn is_valid(&self, now: Timestamp) -> bool {
 224          let mut prev_to: Option<PrincipalId> = None;
 225  
 226          for (_i, link) in self.links.iter().enumerate() {
 227              // Check expiry
 228              if link.expires <= now {
 229                  return false;
 230              }
 231  
 232              // Check continuity: grantor must be the delegatee of the previous link
 233              if let Some(prev) = prev_to {
 234                  if link.from != prev {
 235                      return false;
 236                  }
 237              }
 238              prev_to = Some(link.to);
 239  
 240              // Verify signature
 241              if !link.verify() {
 242                  return false;
 243              }
 244          }
 245          true
 246      }
 247  }

 249  impl DelegationLink {
 250      /// Verify the signature of this delegation link
 251      pub fn verify(&self) -> bool {
 252          let public_key_bytes = self.from.as_bytes();
 253          let verifying_key = match VerifyingKey::from_bytes(public_key_bytes) {
 254              Ok(k) => k,
 255              Err(_) => return false,
 256          };
 257  
 258          let signature = match Signature::from_slice(&self.signature) {
 259              Ok(s) => s,
 260              Err(_) => return false,
 261          };
 262  
 263          // Create data to verify (canonical representation)
 264          let data = self.signing_data();
 265          verifying_key.verify(&data, &signature).is_ok()
 266      }
 267  
 268      /// Sign the delegation link
 269      pub fn sign(&mut self, signing_key: &SigningKey) -> Result<(), String> {
 270          if signing_key.verifying_key().to_bytes() != *self.from.as_bytes() {
 271              return Err("Signing key does not match grantor PrincipalId".to_string());
 272          }
 273  
 274          let data = self.signing_data();
 275          let signature = signing_key.sign(&data);
 276          self.signature = signature.to_bytes().to_vec();
 277          Ok(())
 278      }
 279  
 280      fn signing_data(&self) -> Vec<u8> {
 281          let mut data = Vec::new();
 282          data.extend_from_slice(self.from.as_bytes());
 283          data.extend_from_slice(self.to.as_bytes());
 284          // For capabilities and expires, use bincode for stable serialization
 285          if let Ok(ser) = bincode::serialize(&(&self.capabilities, self.expires)) {
 286              data.extend_from_slice(&ser);
 287          }
 288          data
 289      }
