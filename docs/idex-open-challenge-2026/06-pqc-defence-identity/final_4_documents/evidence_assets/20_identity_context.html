Embedded Identity Context

PCU identity carries principal, capabilities, delegation, expiry, and signature.
 302  /// Complete identity context for a computation
 303  /// 
 304  /// Key insight: This is embedded IN the PCU, not checked externally.
 305  /// The computation carries proof of authorization.
 306  #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
 307  pub struct IdentityContext {
 308      /// Who requested this computation
 309      pub principal: PrincipalId,
 310      
 311      /// What permissions they have (capability-based)
 312      pub capabilities: CapabilitySet,
 313      
 314      /// Delegation chain (if acting on behalf of another)
 315      pub delegation: Option<DelegationChain>,
 316      
 317      /// Expiry (computation becomes invalid after this)
 318      pub valid_until: Timestamp,
 319      
 320      /// Signature proving identity owns this context
 321      pub signature: Vec<u8>,
 322  }
 323  
 324  impl IdentityContext {
 325      /// Create new identity context (needs to be signed afterwards)
 326      pub fn new(principal: PrincipalId, capabilities: CapabilitySet) -> Self {
 327          IdentityContext {
 328              principal,
 329              capabilities,
 330              delegation: None,
 331              valid_until: crate::now() + 3600_000, // 1 hour default
 332              signature: Vec::new(),
 333          }
 334      }
 335  
 336      /// Create with delegation
 337      pub fn with_delegation(mut self, chain: DelegationChain) -> Self {
 338          self.delegation = Some(chain);
 339          self
 340      }
 341  
 342      /// Create with custom expiry
 343      pub fn with_expiry(mut self, valid_until: Timestamp) -> Self {
 344          self.valid_until = valid_until;
 345          self
 346      }
 347  
 348      /// Check if still valid
 349      pub fn is_valid(&self) -> bool {
 350          let now = crate::now();
 351          
 352          // Check expiry
 353          if now >= self.valid_until {
 354              return false;
 355          }
 356          
 357          // Check delegation chain if present
 358          if let Some(ref chain) = self.delegation {
 359              if !chain.is_valid(now) {
 360                  return false;
 361              }
 362              // If delegated, the context principal must be the final delegatee in the chain
 363              if let Some(final_delegatee) = chain.final_delegatee() {
 364                  if self.principal != final_delegatee {
 365                      return false;
 366                  }
 367              }
 368          }
 369          
 370          // Check signature
 371          if !self.verify() {
 372              return false;
 373          }
 374  
 375          true
 376      }
 377  
 378      /// Verify signature of the context
 379      pub fn verify(&self) -> bool {
 380          let public_key_bytes = self.principal.as_bytes();
 381          let verifying_key = match VerifyingKey::from_bytes(public_key_bytes) {
 382              Ok(k) => k,
 383              Err(_) => return false,
 384          };
 385  
 386          let signature = match Signature::from_slice(&self.signature) {
 387              Ok(s) => s,
 388              Err(_) => return false,
 389          };
 390  
 391          let data = self.signing_data();
 392          verifying_key.verify(&data, &signature).is_ok()
 393      }
 394  
 395      /// Sign the identity context
 396      pub fn sign(&mut self, signing_key: &SigningKey) -> Result<(), String> {
 397          if signing_key.verifying_key().to_bytes() != *self.principal.as_bytes() {
 398              return Err("Signing key does not match principal PrincipalId".to_string());
 399          }
 400  
 401          let data = self.signing_data();
 402          let signature = signing_key.sign(&data);
 403          self.signature = signature.to_bytes().to_vec();
 404          Ok(())
 405      }
 406  
 407      fn signing_data(&self) -> Vec<u8> {
 408          let mut data = Vec::new();
 409          data.extend_from_slice(self.principal.as_bytes());
 410          // Use bincode for stable serialization of complex types
 411          if let Ok(ser) = bincode::serialize(&(&self.capabilities, &self.delegation, self.valid_until)) {
 412              data.extend_from_slice(&ser);
 413          }
 414          data
 415      }
 416  
 417      /// Check if identity permits an action
 418      pub fn permits(&self, resource: &str, action: &str) -> bool {
 419          self.is_valid() && self.capabilities.permits(resource, action)
 420      }
