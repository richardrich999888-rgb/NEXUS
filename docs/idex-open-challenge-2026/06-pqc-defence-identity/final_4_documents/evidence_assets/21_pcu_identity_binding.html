Portable Computation Unit embeds identity context and execution constraints.
 113  /// Portable Computation Unit
 114  /// 
 115  /// A self-contained computation that includes:
 116  /// - Code: The function to execute (WASM bytecode)
 117  /// - Data references: Content-addressed pointers to required data
 118  /// - Identity context: Who requested this, with what permissions
 119  /// - Execution constraints: Where/how this can execute
 120  /// - Execution proof: After execution, cryptographic proof of correctness
 121  ///
 122  /// Key innovation: PCU routes TO where data lives, not the other way around.
 123  #[derive(Debug, Clone, Serialize, Deserialize)]
 124  pub struct PCU {
 125      /// Unique identifier (hash of code + inputs + identity)
 126      pub id: ContentHash,
 127      
 128      /// The WASM module to execute
 129      pub code: WasmModule,
 130      
 131      /// Content-addressed references to input data
 132      /// PCU routes to node where these are locally available
 133      pub inputs: Vec<ContentHash>,
 134      
 135      /// Additional input parameters (inline, not content-addressed)
 136      pub parameters: Vec<u8>,
 137      
 138      /// Identity context (intrinsic, not external)
 139      pub identity: IdentityContext,
 140      
 141      /// Execution constraints
 142      pub constraints: ExecutionConstraints,
 143      
 144      /// When this PCU was created
 145      pub created_at: Timestamp,
 146      
 147      /// Execution result (populated after execution)
 148      pub result: Option<PCUResult>,
 149  }
