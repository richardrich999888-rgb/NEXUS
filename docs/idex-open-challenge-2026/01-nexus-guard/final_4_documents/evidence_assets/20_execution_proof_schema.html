  90  /// Proof of correct execution.
  91  #[derive(Debug, Clone, Serialize, Deserialize)]
  92  pub struct ExecutionProof {
  93      /// Hash of the PCU code executed.
  94      pub pcu_hash: ContentHash,
  95      /// Hashes of the inputs used for execution.
  96      pub input_hashes: Vec<ContentHash>,
  97      /// Hash of the execution output.
  98      pub output_hash: ContentHash,
  99      /// Hash of the identity context used.
 100      pub identity_hash: ContentHash,
 101      /// ID of the node that performed the execution.
 102      pub executor_node: NodeId,
 103      /// Unix timestamp of execution.
 104      pub executed_at: u64,
 105      /// Duration of execution in milliseconds.
 106      pub duration_ms: u64,
 107      /// Fuel consumed during execution.
 108      pub fuel_consumed: u64,
 109      /// Peak memory usage in bytes.
 110      pub peak_memory: usize,
 111      /// Node attestation (signature + optional TEE quote).
 112      pub attestation: NodeAttestation,
 113  }
 114  
 115  impl ExecutionProof {
 116      /// Create a new execution proof.
 117      pub fn create(
 118          pcu: &PCU,
 119          inputs: &[(ContentHash, Vec<u8>)],
 120          result: &ExecutionResult,
 121          identity: &IdentityContext,
 122          node_key: &SigningKey,
 123      ) -> Self {
 124          let pcu_hash = pcu.content_hash();
 125          let input_hashes: Vec<_> = inputs.iter().map(|(h, _)| *h).collect();
 126          let identity_hash = identity.content_hash();
 127          let executor_node = NodeId::from_verifying_key(&node_key.verifying_key());
 128  
 129          let executed_at = SystemTime::now()
 130              .duration_since(UNIX_EPOCH)
 131              .unwrap_or(std::time::Duration::ZERO)
 132              .as_secs();
 133  
 134          let mut proof = Self {
 135              pcu_hash,
 136              input_hashes,
 137              output_hash: result.output_hash,
 138              identity_hash,
 139              executor_node,
 140              executed_at,
 141              duration_ms: result.duration.as_millis() as u64,
 142              fuel_consumed: result.fuel_consumed,
 143              peak_memory: result.peak_memory,
 144              attestation: NodeAttestation::new(node_key.verifying_key().to_bytes()),
 145          };
 146  
 147          let signing_bytes = proof.signing_bytes();
 148          proof.attestation.signature = node_key.sign(&signing_bytes);
 149  
 150          proof
 151      }
 152  
 153      /// Get the bytes to be signed for this proof.
 154      pub fn signing_bytes(&self) -> Vec<u8> {
 155          let mut bytes = Vec::with_capacity(256);
 156          bytes.extend_from_slice(self.pcu_hash.as_bytes());
 157          bytes.extend_from_slice(&(self.input_hashes.len() as u32).to_le_bytes());
 158          for hash in &self.input_hashes {
 159              bytes.extend_from_slice(hash.as_bytes());
 160          }
 161          bytes.extend_from_slice(self.output_hash.as_bytes());
 162          bytes.extend_from_slice(self.identity_hash.as_bytes());
 163          bytes.extend_from_slice(&self.executor_node.0);
 164          bytes.extend_from_slice(&self.executed_at.to_le_bytes());
 165          bytes.extend_from_slice(&self.duration_ms.to_le_bytes());
 166          bytes.extend_from_slice(&self.fuel_consumed.to_le_bytes());
 167          bytes.extend_from_slice(&(self.peak_memory as u64).to_le_bytes());
 168          bytes.extend_from_slice(&self.attestation.node_pubkey);
 169          bytes
 170      }
 171  
 172      /// Verify the proof signature.
 173      pub fn verify(&self) -> Result<(), ProofError> {
 174          let signing_bytes = self.signing_bytes();
 175          self.attestation
 176              .verify(&signing_bytes)
 177              .map_err(|_| ProofError::InvalidSignature)
 178      }
