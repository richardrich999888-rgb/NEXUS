   9  /// Context for PCU execution.
  10  ///
  11  /// Contains all data needed to execute a PCU:

  13  /// - Identity for capability checking

  15  #[derive(Debug, Clone)]
  16  pub struct ExecutionContext {
  17      /// Input data keyed by content hash.
  18      pub inputs: Vec<(ContentHash, Vec<u8>)>,
  19  
  20      /// Identity context for capability verification.
  21      pub identity: IdentityContext,
  22  
  23      /// Resource limits.
  24      pub limits: ExecutionLimits,
  25  
  26      /// Optional: Request ID for tracing.
  27      pub request_id: Option<String>,
  28  
  29      /// Optional: Biological / AHES-derived risk [0, 1] for guard (e.g. stress, cortisol).
  30      /// When set, NervousSystemGuard uses this as estimated_risk for autonomic check.
  31      /// Invariant: write-once per request; not exposed to executing PCU (guest has no access).
  32      pub biological_risk: Option<f64>,
  33  }

  35  impl ExecutionContext {

  41      ) -> Self {
  42          Self {
  43              inputs,
  44              identity,
  45              limits,
  46              request_id: None,
