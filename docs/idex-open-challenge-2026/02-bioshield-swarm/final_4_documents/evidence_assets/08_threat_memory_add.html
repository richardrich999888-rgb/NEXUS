  71      /// Adds a threat report, aggregating with existing reports if pattern matches.
  72      pub fn add(
  73          &mut self,
  74          report: SignedThreatReport,
  75          reputation: &ReputationAggregator,
  76          self_id: AsiId,
  77          current_time: u64,
  78      ) -> ThreatAddResult {
  79          // Check for exact duplicate
  80          let report_id = report.report_id();
  81          if self.threats.contains_key(&report_id) {
  82              return ThreatAddResult::Duplicate;
  83          }
  84          
  85          // Get reporter's reputation
  86          let reporter_rep = reputation.get_direct(self_id, report.reporter, current_time);
  87          
  88          // Reject if reputation too low
  89          const MIN_REPUTATION: f64 = 0.2;
  90          if reporter_rep < MIN_REPUTATION {
  91              return ThreatAddResult::Rejected {
  92                  reason: format!("Reporter reputation {:.2} below threshold", reporter_rep),
  93              };
  94          }
  95          
  96          // Check if this pattern already exists
  97          let pattern_hash = report.pattern.pattern_hash;
  98          if let Some(existing_ids) = self.by_pattern.get(&pattern_hash) {
  99              // Find the primary report for this pattern
 100              if let Some(primary_id) = existing_ids.iter().next() {
 101                  if let Some(stored) = self.threats.get_mut(primary_id) {
 102                      // Don't count same reporter twice
 103                      if stored.confirming_reporters.contains(&report.reporter) {
 104                          return ThreatAddResult::Duplicate;
 105                      }
 106                      
 107                      // Aggregate confidence
 108                      stored.confirming_reporters.insert(report.reporter);
 109                      
 110                      // Confidence increases with more reporters, weighted by reputation
 111                      let weighted_confidence = report.confidence * reporter_rep;
 112                      let n = stored.confirming_reporters.len() as f64;
 113                      stored.aggregated_confidence =
 114                          (stored.aggregated_confidence * (n - 1.0) + weighted_confidence) / n;
 115                      
 116                      // Cap at 1.0
 117                      stored.aggregated_confidence = stored.aggregated_confidence.min(1.0);
 118                      
 119                      return ThreatAddResult::Confirmed {
 120                          new_confidence: stored.aggregated_confidence,
 121                      };
 122                  }
 123              }
 124          }
 125          
 126          // New threat - check capacity
 127          if self.threats.len() >= self.capacity {
 128              self.evict_oldest(current_time);
 129          }
 130          
 131          // Add new threat
 132          let mut confirming = HashSet::new();
 133          confirming.insert(report.reporter);
 134          
 135          let category = report.pattern.category;
 136          let stored = StoredThreat {
 137              aggregated_confidence: report.confidence * reporter_rep,
 138              confirming_reporters: confirming,
 139              first_seen: current_time,
 140              report,
 141          };
 142          
 143          self.threats.insert(report_id, stored);
 144          
 145          self.by_category
 146              .entry(category)
 147              .or_default()
 148              .insert(report_id);
 149          
 150          self.by_pattern
 151              .entry(pattern_hash)
 152              .or_default()
 153              .insert(report_id);
 154          
 155          ThreatAddResult::Added
 156      }
