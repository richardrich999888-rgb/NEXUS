  64      pub fn get(&self, current_time: u64) -> f64 {
  65          let elapsed = current_time.saturating_sub(self.last_update);
  66          let decay = Self::DECAY_RATE.powi(elapsed as i32);
  67          
  68          // Decay toward initial value, not toward zero
  69          Self::INITIAL + (self.value - Self::INITIAL) * decay
  70      }
