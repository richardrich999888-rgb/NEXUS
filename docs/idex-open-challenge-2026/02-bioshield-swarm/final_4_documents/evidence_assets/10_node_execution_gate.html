 255      /// Denies if the principal is isolated due to defection or if aggregated reputation is below threshold.
 256      pub fn allow_execution_by(
 257          &mut self,
 258          principal_id: [u8; 32],
 259          min_reputation: f64,
 260      ) -> Result<(), String> {
 261          let asi_id = AsiId::from_bytes(principal_id);

 264          }
 265          let rep = self.reputation.get_aggregated(self.identity.id, asi_id, self.current_time);
 266          if rep < min_reputation {
 267              return Err(format!(
 268                  "Reputation {} below threshold {}",
 269                  rep, min_reputation
 270              ));
 271          }
 272          Ok(())
 273      }
