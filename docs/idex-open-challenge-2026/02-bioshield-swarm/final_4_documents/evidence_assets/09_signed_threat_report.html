Threat reports bind pattern, reporter, confidence, timestamp, and signature.
  10  /// A signed threat report from an ASI.
  11  #[derive(Debug, Clone)]
  12  pub struct SignedThreatReport {
  13      /// The threat pattern being reported.
  14      pub pattern: ThreatPattern,
  15      /// ID of the reporting ASI.
  16      pub reporter: AsiId,
  17      /// Confidence in this threat [0, 1].
  18      pub confidence: f64,
  19      /// Monotonic timestamp.
  20      pub timestamp: u64,
  21      /// Cryptographic signature over (pattern, reporter, confidence, timestamp).
  22      pub signature: Signature,
  23  }
  24  
  25  impl SignedThreatReport {
  26      /// Creates and signs a new threat report.
  27      pub fn new(
  28          identity: &AsiIdentity,
  29          pattern: ThreatPattern,
  30          confidence: f64,
  31          timestamp: u64,
  32      ) -> Self {
  33          let confidence = confidence.clamp(0.0, 1.0);
  34          let signature = Self::sign_report(identity, &pattern, confidence, timestamp);
  35          
  36          Self {
  37              pattern,
  38              reporter: identity.id,
  39              confidence,
  40              timestamp,
  41              signature,
  42          }
  43      }
  44      
  45      fn sign_report(
  46          identity: &AsiIdentity,
  47          pattern: &ThreatPattern,
  48          confidence: f64,
  49          timestamp: u64,
  50      ) -> Signature {
  51          let message = Self::build_message(pattern, identity.id, confidence, timestamp);
  52          identity.sign(&message)
  53      }
  54      
  55      fn build_message(
  56          pattern: &ThreatPattern,
  57          reporter: AsiId,
  58          confidence: f64,
  59          timestamp: u64,
  60      ) -> Vec<u8> {
  61          let mut hasher = Sha256::new();
  62          hasher.update(b"THREAT_REPORT_V1:");
  63          hasher.update(&pattern.pattern_hash);
  64          hasher.update(&[pattern.category as u8]);
  65          hasher.update(&pattern.severity.to_le_bytes());
  66          hasher.update(&reporter.0);
  67          hasher.update(&confidence.to_le_bytes());
  68          hasher.update(&timestamp.to_le_bytes());
  69          hasher.finalize().to_vec()
  70      }
  71      
  72      /// Verifies the signature on this report.
  73      pub fn verify(&self, public_identity: &PublicIdentity) -> bool {
  74          if public_identity.id != self.reporter {
  75              return false;
  76          }
  77          
  78          let message = Self::build_message(
  79              &self.pattern,
  80              self.reporter,
  81              self.confidence,
  82              self.timestamp,
  83          );
  84          
  85          public_identity.verify(&message, &self.signature)
  86      }
  87      
  88      /// Computes unique identifier for deduplication.
  89      pub fn report_id(&self) -> [u8; 32] {
  90          let mut hasher = Sha256::new();
  91          hasher.update(&self.reporter.0);
  92          hasher.update(&self.pattern.pattern_hash);
  93          hasher.update(&self.timestamp.to_le_bytes());
  94          
  95          let hash = hasher.finalize();
  96          let mut id = [0u8; 32];
  97          id.copy_from_slice(&hash);
  98          id
  99      }

 101      /// Returns the effective severity (pattern severity * confidence).
 102      pub fn effective_severity(&self) -> f64 {
 103          self.pattern.effective_severity() * self.confidence
 104      }
