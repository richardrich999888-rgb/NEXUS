 202      def simulate_sensor(self, robot_id: str, sensor_type: str, data: Dict):
 203          """Inject simulated sensor data"""
 204          if robot_id not in self.robots:
 205              return {"error": "Robot not found"}
 206          
 207          robot = self.robots[robot_id]
 208          robot.sensors[sensor_type] = data
 209          
 210          # Trigger callback if subscribed
 211          topic = f"{robot_id}/scan" if sensor_type == "lidar" else f"{robot_id}/{sensor_type}"
 212          if topic in self.topics and self.topics[topic].callback:
 213              self.topics[topic].callback(data)
 214              self.topics[topic].receive_count += 1
 215          
 216          return {"status": "simulated", "sensor": sensor_type}

 218      def get_stats(self) -> Dict:
 219          """Get bridge statistics"""
 220          return {
 221              "topics": len(self.topics),
 222              "robots": len(self.robots),
 223              "publishers": sum(1 for t in self.topics.values() if t.is_publisher),
 224              "subscribers": sum(1 for t in self.topics.values() if not t.is_publisher),
 225              "total_published": sum(t.publish_count for t in self.topics.values()),
 226              "total_received": sum(t.receive_count for t in self.topics.values())
