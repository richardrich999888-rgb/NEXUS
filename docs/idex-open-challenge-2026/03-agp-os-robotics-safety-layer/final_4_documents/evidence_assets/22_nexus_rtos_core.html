  64  pub struct StaticScheduler<const N: usize> {
  65      tasks: [Option<RtTask>; N],
  66      queued: usize,
  67      tick: u64,

  71  impl<const N: usize> StaticScheduler<N> {
  72      /// Create an empty scheduler.
  73      pub const fn new() -> Self {
  74          Self {
  75              tasks: [None; N],
  76              queued: 0,
  77              tick: 0,
  78              missed_deadlines: 0,
  79          }
  80      }

  82      /// Queue a task.
  83      pub fn submit(&mut self, task: RtTask) -> Result<(), SchedulerError> {
  84          if self
  85              .tasks
  86              .iter()
  87              .flatten()
  88              .any(|queued| queued.id == task.id)
  89          {
  90              return Err(SchedulerError::DuplicateTaskId);
  91          }
  92  
  93          for slot in &mut self.tasks {
  94              if slot.is_none() {
  95                  *slot = Some(task);
  96                  self.queued += 1;
  97                  return Ok(());
  98              }
  99          }
 100  
 101          Err(SchedulerError::Full)
 102      }
 103  
 104      /// Return the next task without removing it.
 105      pub fn peek_next(&self) -> Option<RtTask> {
 106          self.best_slot().and_then(|index| self.tasks[index])
 107      }
 108  
 109      /// Pop the next task by priority, then earliest deadline, then stable id.
 110      pub fn pop_next(&mut self) -> Option<RtTask> {
 111          let index = self.best_slot()?;
 112          let task = self.tasks[index].take()?;
 113          self.queued -= 1;
 114          if self.tick > task.deadline_ticks {
 115              self.missed_deadlines = self.missed_deadlines.saturating_add(1);
 116          }
 117          Some(task)
 118      }
